cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(outlands_pyexpat_update LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(EXPAT_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(EXPAT_MSVC_STATIC_CRT OFF CACHE BOOL "" FORCE)
add_subdirectory("${EXPAT_SOURCE}" expat)
add_library(pyexpat MODULE "${CPY_SOURCE}/Modules/pyexpat.c")
target_include_directories(pyexpat PRIVATE "${PY_INCLUDE}" "${PY_INCLUDE}/internal" "${CPY_SOURCE}/Include" "${CPY_SOURCE}/Include/internal" "${CPY_SOURCE}/Modules" "${EXPAT_SOURCE}/lib" "${CMAKE_CURRENT_BINARY_DIR}/expat")
target_compile_definitions(pyexpat PRIVATE XML_STATIC _CRT_SECURE_NO_WARNINGS)
target_link_libraries(pyexpat PRIVATE expat)
set_target_properties(pyexpat PROPERTIES PREFIX "")
if(WIN32)
  target_link_libraries(pyexpat PRIVATE "${PY_ROOT}/libs/python${PY_DIGITS}.lib")
  set_target_properties(pyexpat PROPERTIES SUFFIX ".pyd")
else()
  set_target_properties(pyexpat PROPERTIES SUFFIX "${PY_SUFFIX}")
endif()
